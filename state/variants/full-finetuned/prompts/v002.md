function amicable_numbers_sum(limit) {
    let sum = 0;
    for (let i = 1; i <= limit; i++) {
        let divisorsSum = 0;
        for (let j = 1; j < i; j++) {
            if (i % j === 0) {
                divisorsSum += j;
            }
        }
        if (divisorsSum === i && i !== sum) {
            sum += i;
        }
    }
    return sum;
}
